-- | Una mónada sobre una categoría es un monoide en sus endofuntores.
class Applicative m => Monad m where
  (>>=)  :: m a -> (a -> m b) -> m b
  return :: a -> m a
  return = pure

main :: IO ()
main = mapM_ print [x * x | x <- [1 .. 5 :: Int]]

Python:

def fib(n: int) -> int:
    """Fibonacci, en versión iterativa."""
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a
